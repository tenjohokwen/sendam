Create src/main/resources/db/migration/V2__client_api_key.sql with exactly this content:

```sql
CREATE TABLE main.client_account (
    id                  BIGINT PRIMARY KEY,
    name                VARCHAR(100) NOT NULL,
    status              VARCHAR(20)  NOT NULL DEFAULT 'ACTIVE',
    created_by          VARCHAR(50),
    created_date        TIMESTAMP,
    last_modified_by    VARCHAR(50),
    last_modified_date  TIMESTAMP,
    request_id          VARCHAR(100),
    session_id          TEXT
);

CREATE TABLE main.client_api_key (
    id                  BIGINT PRIMARY KEY,
    client_id           BIGINT NOT NULL REFERENCES main.client_account(id),
    key_prefix          VARCHAR(24) NOT NULL UNIQUE,
    key_hash            VARCHAR(64) NOT NULL,
    label               VARCHAR(100),
    status              VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
    created_by          VARCHAR(50),
    created_date        TIMESTAMP,
    last_modified_by    VARCHAR(50),
    last_modified_date  TIMESTAMP,
    request_id          VARCHAR(100),
    session_id          TEXT
);

CREATE INDEX idx_client_api_key_prefix ON main.client_api_key(key_prefix);
```

Column layout mirrors the pattern in `AbstractAuditingEntity` (created_by, created_date, last_modified_by, last_modified_date, request_id, session_id, status). The `id` column in both tables is a BIGINT populated by the `@Tsid` generator (not a DB-generated sequence). `key_prefix` stores "snd_" + 16 hex chars (20 chars max, VARCHAR(24) for safety). `key_hash` is HMAC-SHA256 hex output (always 64 chars). Do NOT add a `version` column — `AbstractAuditingEntity` does not use optimistic locking in this codebase.

Flyway config in application.yaml: `defaultSchema: main`, `baseline-on-migrate: true`. This is V2 — check src/main/resources/db/migration/ to confirm no V2 already exists before creating (it should be empty or contain only V1).

  </verify>

```java
@Service
@Transactional
public class ClientService {
    private final ClientRepository clientRepository;
    private final ClientApiKeyRepository apiKeyRepository;

    // constructor injection

    public CreateClientResponse createClient(CreateClientRequest request) {
        ClientEntity client = ClientEntity.builder()
            .name(request.name())
            .status(EntityStatus.ACTIVE)
            .build();
        clientRepository.save(client);

        // Stub: Plan 02 replaces this with ApiKeyService.generateAndPersist(clientId, label)
        String rawKey = "STUB_REPLACE_IN_PLAN_02";
        ClientApiKeyEntity keyEntity = ClientApiKeyEntity.builder()
            .clientId(client.getId())
            .keyPrefix("snd_stub_prefix")
            .keyHash("stub_hash")
            .label(request.keyLabel())
            .status(EntityStatus.ACTIVE)
            .build();
        apiKeyRepository.save(keyEntity);

        return new CreateClientResponse(client.getId(), keyEntity.getId(), rawKey);
    }
}
```
