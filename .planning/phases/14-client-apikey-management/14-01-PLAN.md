**api/admin/index.js** — Replace the existing placeholder file entirely with:

```js
import { api } from 'src/boot/axios'

export const adminApi = {
  getClients() {
    return api.get('/api/admin/clients')
  },
  createClient(data) {
    return api.post('/api/admin/clients', data)
  },
  getApiKeys(clientId) {
    return api.get(`/api/admin/clients/${clientId}/keys`)
  },
  createApiKey(clientId, label) {
    return api.post(
      `/api/admin/clients/${clientId}/keys`,
      label ? { label } : undefined
    )
  },
  revokeApiKey(clientId, keyId) {
    return api.delete(`/api/admin/clients/${clientId}/keys/${keyId}`)
  }
}
```

Note: `clientId` and `keyId` must already be strings (longToString applied before calling these functions). The axios interceptor in boot/axios.js unwraps `response.data` — callers receive the DTO directly.

**src/stores/user.store.js** — Create new file:

```js
import { defineStore } from 'pinia'
import { profileApi } from 'src/api/profile'

export const useUserStore = defineStore('user', {
  state: () => ({
    authorities: [],
    username: ''
  }),
  getters: {
    isAdmin: (state) => state.authorities.includes('ROLE_ADMIN'),
    isLoaded: (state) => state.authorities.length > 0 || state.username !== ''
  },
  actions: {
    async fetchUser() {
      const user = await profileApi.getProfile()
      this.authorities = user.authorities ?? []
      this.username = user.login ?? user.email ?? ''
    },
    reset() {
      this.authorities = []
      this.username = ''
    }
  }
})
```

Note: `UserDto.authorities` is `Set<String>` in Java, serialized as a JSON array. Values are strings like `"ROLE_ADMIN"`, `"ROLE_USER"`. Check the existing `profileApi` import path by reading `src/api/profile/index.js` or `src/api/profile.js` — use whatever path exists.
