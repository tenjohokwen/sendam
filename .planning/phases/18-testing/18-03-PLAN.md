**CreateClientDialog.test.js**

Component at `src/components/admin/CreateClientDialog.vue`. Key facts:
- Props: `modelValue` (Boolean, required)
- Emits: `update:modelValue`, `created`
- `onSubmit` calls `adminApi.createClient({ name })`, then `$q.notify({ type: 'positive' })`, emits `created`, calls `onClose` (which sets modelValue false)
- `onClose` always resets form and sets `dialogVisible.value = false` → emits `update:modelValue=false`
- useErrorHandler: 400 field errors come through `setError(err)` which sets `isValidationError` and makes `hasFieldError('name')` truthy when backend returns field-level errors

```javascript
import { installQuasarPlugin } from '@quasar/quasar-app-extension-testing-unit-vitest'
import { mount, flushPromises } from '@vue/test-utils'
import { createI18n } from 'vue-i18n'
import { vi, describe, it, expect, beforeEach } from 'vitest'
import messages from 'src/i18n'
import { adminApi } from 'src/api/admin'
import CreateClientDialog from 'src/components/admin/CreateClientDialog.vue'

vi.mock('src/api/admin', () => ({
  adminApi: {
    createClient: vi.fn(),
  },
}))

installQuasarPlugin({ plugins: { Notify: (await import('quasar')).Notify } })

const i18n = createI18n({ locale: 'en-US', legacy: false, globalInjection: true, messages })

function mountComponent(modelValue = true) {
  const wrapper = mount(CreateClientDialog, {
    global: { plugins: [i18n] },
    props: { modelValue },
  })
  wrapper.vm.$q.notify = vi.fn()
  return wrapper
}

describe('CreateClientDialog', () => {
  beforeEach(() => {
    vi.clearAllMocks()
  })

  it('calls adminApi.createClient with name on submit', async () => {
    adminApi.createClient.mockResolvedValue({})
    const wrapper = mountComponent()
    await wrapper.find('input').setValue('Test Client')
    await wrapper.find('form').trigger('submit')
    await flushPromises()
    expect(adminApi.createClient).toHaveBeenCalledWith(
      expect.objectContaining({ name: 'Test Client' })
    )
  })

  it('emits created event on successful submit', async () => {
    adminApi.createClient.mockResolvedValue({})
    const wrapper = mountComponent()
    await wrapper.find('input').setValue('Test Client')
    await wrapper.find('form').trigger('submit')
    await flushPromises()
    expect(wrapper.emitted('created')).toBeTruthy()
  })

  it('emits update:modelValue=false on successful submit (dialog closes)', async () => {
    adminApi.createClient.mockResolvedValue({})
    const wrapper = mountComponent()
    await wrapper.find('input').setValue('My Client')
    await wrapper.find('form').trigger('submit')
    await flushPromises()
    const emitted = wrapper.emitted('update:modelValue')
    expect(emitted).toBeTruthy()
    // Last emit should be false (close)
    const lastEmit = emitted[emitted.length - 1]
    expect(lastEmit).toEqual([false])
  })

  it('does not call createClient and emits update:modelValue=false on cancel', async () => {
    const wrapper = mountComponent()
    // Click the cancel button (flat button with 'cancel' label)
    const cancelBtn = wrapper.findAll('.q-btn').find(b => b.text().toLowerCase().includes('cancel'))
    await cancelBtn.trigger('click')
    expect(adminApi.createClient).not.toHaveBeenCalled()
    expect(wrapper.emitted('update:modelValue')).toBeTruthy()
  })

  it('shows error banner on API failure (non-validation error)', async () => {
    const networkError = new Error('Network Error')
    adminApi.createClient.mockRejectedValue(networkError)
    const wrapper = mountComponent()
    await wrapper.find('input').setValue('Client Name')
    await wrapper.find('form').trigger('submit')
    await flushPromises()
    expect(wrapper.find('.q-banner').exists()).toBe(true)
  })
})
```

Create `src/frontend/test/vitest/__tests__/components/admin/DlrDialog.test.js`.

Component at `src/components/admin/DlrDialog.vue`. Key facts:
- Props: `show` (Boolean), `sendRequestId` (String|null)
- Emits: `update:show`
- CRITICAL: `watch(() => props.sendRequestId, ..., { immediate: true })` — `loadDlr()` is called synchronously during component creation when `sendRequestId` is non-null. Mock `getDlrForRequest` BEFORE mounting with a non-null sendRequestId.
- `loadDlr()` calls `adminApi.getDlrForRequest(sendRequestId, { page: currentPage - 1, size: 20 })`
- currentPage starts at 1, so first call uses `{ page: 0, size: 20 }`
- ServerPagination rendered when `totalPages > 1`
- `onPageChange(page)` sets `currentPage.value = page` then calls `loadDlr()`

```javascript
import { installQuasarPlugin } from '@quasar/quasar-app-extension-testing-unit-vitest'
import { mount, flushPromises } from '@vue/test-utils'
import { createI18n } from 'vue-i18n'
import { vi, describe, it, expect, beforeEach } from 'vitest'
import messages from 'src/i18n'
import { adminApi } from 'src/api/admin'
import DlrDialog from 'src/components/admin/DlrDialog.vue'
import ServerPagination from 'src/components/common/ServerPagination.vue'

vi.mock('src/api/admin', () => ({
  adminApi: {
    getDlrForRequest: vi.fn(),
  },
}))

installQuasarPlugin()

const i18n = createI18n({ locale: 'en-US', legacy: false, globalInjection: true, messages })

const emptyPage = { content: [], totalElements: 0, totalPages: 0 }

function mountComponent(props = {}) {
  return mount(DlrDialog, {
    global: { plugins: [i18n] },
    props: {
      show: true,
      sendRequestId: 'req-001',
      ...props,
    },
  })
}

describe('DlrDialog', () => {
  beforeEach(() => {
    vi.clearAllMocks()
  })

  it('calls getDlrForRequest immediately on mount when sendRequestId is set (watch immediate)', async () => {
    // MUST mock BEFORE mount — watch is { immediate: true }
    adminApi.getDlrForRequest.mockResolvedValue(emptyPage)
    mountComponent()
    await flushPromises()
    expect(adminApi.getDlrForRequest).toHaveBeenCalledWith('req-001', { page: 0, size: 20 })
  })

  it('does not call getDlrForRequest when sendRequestId is null', async () => {
    adminApi.getDlrForRequest.mockResolvedValue(emptyPage)
    mountComponent({ sendRequestId: null })
    await flushPromises()
    expect(adminApi.getDlrForRequest).not.toHaveBeenCalled()
  })

  it('renders DLR rows from API response', async () => {
    adminApi.getDlrForRequest.mockResolvedValue({
      content: [
        { id: '1', recipient: '237600000001', sendStatus: 'COMPLETED',
          gatewayMessageId: 'gw-1', providerMessageId: 'prov-1', segmentsConsumed: 1 },
      ],
      totalElements: 1,
      totalPages: 1,
    })
    const wrapper = mountComponent()
    await flushPromises()
    expect(wrapper.text()).toContain('237600000001')
  })

  it('shows error banner when getDlrForRequest rejects', async () => {
    adminApi.getDlrForRequest.mockRejectedValue(new Error('Network Error'))
    const wrapper = mountComponent()
    await flushPromises()
    expect(wrapper.find('.q-banner').exists()).toBe(true)
  })

  it('hides ServerPagination when totalPages <= 1', async () => {
    adminApi.getDlrForRequest.mockResolvedValue(emptyPage)
    const wrapper = mountComponent()
    await flushPromises()
    expect(wrapper.findComponent(ServerPagination).exists()).toBe(false)
  })

  it('shows ServerPagination when totalPages > 1', async () => {
    adminApi.getDlrForRequest.mockResolvedValue({
      content: [], totalElements: 50, totalPages: 3,
    })
    const wrapper = mountComponent()
    await flushPromises()
    expect(wrapper.findComponent(ServerPagination).exists()).toBe(true)
  })

  it('reloads with new sendRequestId when prop changes', async () => {
    adminApi.getDlrForRequest.mockResolvedValue(emptyPage)
    const wrapper = mountComponent({ sendRequestId: 'req-001' })
    await flushPromises()

    adminApi.getDlrForRequest.mockResolvedValue(emptyPage)
    await wrapper.setProps({ sendRequestId: 'req-002' })
    await flushPromises()

    expect(adminApi.getDlrForRequest).toHaveBeenCalledWith('req-002', { page: 0, size: 20 })
  })

  it('calls getDlrForRequest with page 1 (0-based) when ServerPagination emits page-change with 1', async () => {
    adminApi.getDlrForRequest.mockResolvedValue({
      content: [], totalElements: 50, totalPages: 3,
    })
    const wrapper = mountComponent()
    await flushPromises()

    const pagination = wrapper.findComponent(ServerPagination)
    await pagination.vm.$emit('page-change', 1)
    await flushPromises()

    expect(adminApi.getDlrForRequest).toHaveBeenCalledWith('req-001', { page: 1, size: 20 })
  })
})
```
